% KOMA convention: \markboth sets the chapter (left mark) and clears the section (right mark),
% so the start of a chapter never shows the previous chapter's last section or the table of contents.
% \markright sets the section only.
\newcommand{\setchapterfooter}[1]{%
  \def\mydummy{#1}
  \renewcommand{\chaptermark}[1]{%
    \markboth{\textbf{\thechapter}~|~\mydummy}{}}
}

\newcommand{\resetchapterfooter}{%
  \renewcommand{\chaptermark}[1]{%
    \markboth{\textbf{\thechapter}~|~##1}{}}
}

\renewcommand{\chaptermark}[1]{%
  \markboth{\textbf{\thechapter}~|~#1}{}}

\renewcommand{\sectionmark}[1]{%
  \markright{\textbf{\thesection}~|~#1}}

\ofoot[\pagemark]{\pagemark}
\refoot{\leftmark}
\lofoot{\rightmark}
